cmake_minimum_required(VERSION 3.20)
project(Libstone VERSION 0.1.0 LANGUAGES C CXX)
include(cmake/LibstoneBuild.cmake)

if(LIBSTONE_BUILD_TESTS)
  enable_testing()
endif()
# Yellowstone has been moved out of the active tree by the checkout owner.
option(LIBSTONE_BUILD_YELLOWSTONE "Build Yellowstone when present in the active tree" OFF)
if(LIBSTONE_BUILD_YELLOWSTONE AND NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/yellowstone/CMakeLists.txt")
  message(FATAL_ERROR "Yellowstone is not present in the active source tree")
endif()
foreach(component IN ITEMS blackstone firestone limestone sandstone yellowstone tourchstone flintstone)
  string(TOUPPER "${component}" key)
  option(LIBSTONE_BUILD_${key} "Build ${component}" ON)
  if(LIBSTONE_BUILD_${key})
    add_subdirectory(${component})
  endif()
endforeach()
option(LIBSTONE_BUILD_GATESTONE "Build the GateML frontend-definition compiler" OFF)
if(LIBSTONE_BUILD_GATESTONE)
  add_subdirectory(gatestone)
endif()
