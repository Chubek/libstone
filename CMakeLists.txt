cmake_minimum_required(VERSION 3.20)
project(Libstone VERSION 0.1.0 LANGUAGES C CXX)
include(cmake/LibstoneBuild.cmake)

if(LIBSTONE_BUILD_TESTS)
  enable_testing()
endif()
option(LIBSTONE_BUILD_ASDL "Build the native ASDL toolchain" OFF)
if(LIBSTONE_BUILD_ASDL)
  set(ASDL_BUILD_TESTS "${LIBSTONE_BUILD_TESTS}")
  set(ASDL_BUILD_TOOLS "${LIBSTONE_BUILD_TOOLS}")
  add_subdirectory(cornerstone/ASDL)
endif()
option(LIBSTONE_BUILD_AMALGAM8 "Build the Amalgam8 staged header generator" OFF)
if(LIBSTONE_BUILD_AMALGAM8)
  add_subdirectory(cornerstone/Amalgam8)
endif()
option(LIBSTONE_BUILD_AUROCKS "Build the Aurocks parser generator" OFF)
if(LIBSTONE_BUILD_AUROCKS AND LIBSTONE_BUILD_TOOLS)
  add_subdirectory(cornerstone/Aurocks)
endif()
# Keep the native object/linker suite opt-in.
option(LIBSTONE_BUILD_YELLOWSTONE "Build Yellowstone object and linker libraries" OFF)
if(LIBSTONE_BUILD_YELLOWSTONE AND NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/yellowstone/CMakeLists.txt")
  message(FATAL_ERROR "Yellowstone is not present in the active source tree")
endif()
foreach(component IN ITEMS blackstone firestone sandstone limestone yellowstone tourchstone novostone wetstone weighstone moonstone flintstone)
  string(TOUPPER "${component}" key)
  option(LIBSTONE_BUILD_${key} "Build ${component}" ON)
  if(LIBSTONE_BUILD_${key})
    add_subdirectory(${component})
  endif()
endforeach()
option(LIBSTONE_BUILD_GATESTONE "Build the GateML frontend-definition compiler" OFF)
if(LIBSTONE_BUILD_GATESTONE)
  add_subdirectory(gatestone)
endif()
option(LIBSTONE_BUILD_CEPHYR "Build the Cephyr C11 compiler driver" OFF)
if(LIBSTONE_BUILD_CEPHYR)
  add_subdirectory(cornerstone/Cephyr)
endif()
if(LIBSTONE_BUILD_AUROCKS AND NOT TARGET aurocks_compiler)
  add_subdirectory(cornerstone/Aurocks)
endif()
