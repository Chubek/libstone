cmake_minimum_required(VERSION 3.20)
project(FirestoneConsumer LANGUAGES CXX)
find_package(Firestone 0.1 CONFIG REQUIRED COMPONENTS ir rewrite trs eqsat cps memory object sema type vm bignum compute concurrency gfx intrin isel isched regalloc)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE Firestone::firestone)
add_executable(vm_consumer vm.cpp)
target_link_libraries(vm_consumer PRIVATE Firestone::vm)
add_executable(foundation_consumer foundations.cpp)
target_link_libraries(foundation_consumer PRIVATE Firestone::bignum Firestone::compute Firestone::concurrency Firestone::gfx Firestone::intrin)
add_executable(trs_consumer trs.cpp)
target_link_libraries(trs_consumer PRIVATE Firestone::trs)

add_executable(codegen_consumer codegen.cpp)
target_link_libraries(codegen_consumer PRIVATE Firestone::isel Firestone::isched)
add_executable(regalloc_consumer regalloc.cpp)
target_link_libraries(regalloc_consumer PRIVATE Firestone::regalloc)
