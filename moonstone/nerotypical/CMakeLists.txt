cmake_minimum_required(VERSION 3.20)
project(NerotypicalSequent VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
include(GNUInstallDirs)
add_library(nerotypical_sequent INTERFACE)
add_library(Nerotypical::Sequent ALIAS nerotypical_sequent)
target_compile_features(nerotypical_sequent INTERFACE cxx_std_20)
target_include_directories(nerotypical_sequent INTERFACE
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
set_target_properties(nerotypical_sequent PROPERTIES EXPORT_NAME Sequent)
install(TARGETS nerotypical_sequent EXPORT NerotypicalTargets)
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT NerotypicalTargets NAMESPACE Nerotypical:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Nerotypical)
option(NEROTYPICAL_SEQUENT_TESTS "Build Sequent IR tests" ON)
if(BUILD_TESTING AND NEROTYPICAL_SEQUENT_TESTS)
  add_executable(nerotypical_sequent_tests tests/sequent_ir.cpp)
  target_link_libraries(nerotypical_sequent_tests PRIVATE nerotypical_sequent)
  if(MSVC)
    target_compile_options(nerotypical_sequent_tests PRIVATE /UNDEBUG /W4)
  else()
    target_compile_options(nerotypical_sequent_tests PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
  endif()
  add_test(NAME nerotypical.sequent COMMAND nerotypical_sequent_tests)
endif()
