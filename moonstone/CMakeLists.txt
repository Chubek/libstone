  message(FATAL_ERROR "ASan/UBSan and TSan cannot be combined")

if(MOONSTONE_ENABLE_ASAN_UBSAN OR MOONSTONE_ENABLE_TSAN)
  if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Moonstone sanitizers require GCC or Clang with sanitizer runtimes")
  endif()
  if(MOONSTONE_ENABLE_TSAN)
    set(_moonstone_sanitizer -fsanitize=thread)
  else()
    set(_moonstone_sanitizer -fsanitize=address,undefined)
  endif()
endif()

function(moonstone_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endfunction()

add_library(moonstone_core core/Core.cpp core/SourceManager.cpp core/Diagnostics.cpp core/CApi.cpp)
add_library(Moonstone::Core ALIAS moonstone_core)
set_target_properties(moonstone_core PROPERTIES EXPORT_NAME Core VERSION ${PROJECT_VERSION} SOVERSION 1)
target_compile_features(moonstone_core PUBLIC cxx_std_20)
target_include_directories(moonstone_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/moonstone>)
target_link_libraries(moonstone_core PUBLIC Threads::Threads)
target_compile_definitions(moonstone_core PRIVATE MOONSTONE_BUILDING_CORE)
get_target_property(_moonstone_core_type moonstone_core TYPE)
if(_moonstone_core_type STREQUAL "SHARED_LIBRARY")
  target_compile_definitions(moonstone_core PUBLIC MOONSTONE_SHARED)
endif()
if(_moonstone_sanitizer)
  target_compile_options(moonstone_core PUBLIC ${_moonstone_sanitizer} -fno-omit-frame-pointer)
  target_link_options(moonstone_core PUBLIC ${_moonstone_sanitizer})
endif()
moonstone_warnings(moonstone_core)
set(_moonstone_targets moonstone_core)

function(moonstone_component name enabled public_name)
  if(${enabled})
    add_library(moonstone_${name} INTERFACE)
    add_library(Moonstone::${public_name} ALIAS moonstone_${name})
    set_target_properties(moonstone_${name} PROPERTIES EXPORT_NAME ${public_name})
    target_link_libraries(moonstone_${name} INTERFACE moonstone_core)
    target_include_directories(moonstone_${name} INTERFACE
      $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/${name}/include>
      $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/moonstone/${name}/include>)
    set(_moonstone_targets ${_moonstone_targets} moonstone_${name} PARENT_SCOPE)
  endif()
endfunction()
moonstone_component(chisel MOONSTONE_BUILD_CHISEL Chisel)
moonstone_component(syren MOONSTONE_BUILD_SYREN Syren)
moonstone_component(supersonic MOONSTONE_BUILD_SUPERSONIC Supersonic)

if(MOONSTONE_BUILD_TESTS)
  enable_testing()
  function(moonstone_test name src target)
    add_executable(moonstone_test_${name} "${src}")
    target_link_libraries(moonstone_test_${name} PRIVATE ${target})
    if(MSVC)
      target_compile_options(moonstone_test_${name} PRIVATE /UNDEBUG)
    else()
      target_compile_options(moonstone_test_${name} PRIVATE -UNDEBUG)
    endif()
    moonstone_warnings(moonstone_test_${name})
    add_test(NAME moonstone.${name} COMMAND moonstone_test_${name})
  endfunction()
  moonstone_test(core core/tests/core.cpp moonstone_core)
  moonstone_test(c_abi core/tests/c_api.c moonstone_core)
  if(MOONSTONE_BUILD_CHISEL)
    moonstone_test(borrow chisel/tests/borrow.cpp moonstone_chisel)
    moonstone_test(gc chisel/tests/gc.cpp moonstone_chisel)
  endif()
  if(MOONSTONE_BUILD_SUPERSONIC)
    moonstone_test(synthesis supersonic/tests/synthesis.cpp moonstone_supersonic)
    moonstone_test(cegis supersonic/tests/cegis.cpp moonstone_supersonic)
  endif()
  if(MOONSTONE_BUILD_SYREN)
    moonstone_test(catalog syren/tests/catalog.cpp moonstone_syren)
    moonstone_test(repair syren/tests/repair.cpp moonstone_syren)
  endif()
  if(MOONSTONE_BUILD_EXOML)
    moonstone_test(exoml exoml/tests/interpreter.cpp moonstone_exoml)
  endif()
  if(MOONSTONE_BUILD_CHISEL AND MOONSTONE_BUILD_SYREN AND MOONSTONE_BUILD_SUPERSONIC AND MOONSTONE_BUILD_EXOML)
    moonstone_test(integration tests/integration.cpp moonstone_chisel)
    target_link_libraries(moonstone_test_integration PRIVATE moonstone_syren moonstone_supersonic moonstone_exoml)
  endif()

  # Every public header is the first and only include in its translation unit.
  file(GLOB_RECURSE _moonstone_headers RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
    "core/*.hpp" "core/*.h" "chisel/include/*.hpp" "syren/include/*.hpp"
    "supersonic/include/*.hpp" "exoml/include/*.hpp" "integrations/*.hpp")

    string(MAKE_C_IDENTIFIER "${header}" header_id)
    set(src "${CMAKE_CURRENT_BINARY_DIR}/header-checks/${header_id}.cpp")

    add_library(moonstone_header_${header_id} OBJECT "${src}")
    target_link_libraries(moonstone_header_${header_id} PRIVATE moonstone_core)
    moonstone_warnings(moonstone_header_${header_id})
  endforeach()
  add_test(NAME moonstone.configure_contract

    COMMAND ${CMAKE_COMMAND} -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}
      -DBUILD_DIR=${CMAKE_BINARY_DIR} -DCONFIG=$<CONFIG>
      -DTEST_DIR=${CMAKE_CURRENT_BINARY_DIR}/package-test
      -DC_COMPILER=${CMAKE_C_COMPILER} -DCXX_COMPILER=${CMAKE_CXX_COMPILER}
      -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/TestPackage.cmake)
