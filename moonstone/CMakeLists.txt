target_include_directories(moonstone_core INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)

function(moonstone_component name enabled)
  if(${enabled})
    add_library(moonstone_${name} INTERFACE)
    string(SUBSTRING "${name}" 0 1 first)
    string(TOUPPER "${first}" first_upper)
    string(SUBSTRING "${name}" 1 -1 rest)
    add_library(Moonstone::${first_upper}${rest} ALIAS moonstone_${name})
    target_link_libraries(moonstone_${name} INTERFACE moonstone_core)
    target_include_directories(moonstone_${name} INTERFACE
      $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/${name}/include>
      $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
  endif()
endfunction()

if(MOONSTONE_BUILD_TESTS)
  enable_testing()
  function(moonstone_test name src target)
    add_executable(moonstone_test_${name} "${src}")
    target_link_libraries(moonstone_test_${name} PRIVATE ${target})
    if(MSVC)
      target_compile_options(moonstone_test_${name} PRIVATE /UNDEBUG /W4)
    else()
      target_compile_options(moonstone_test_${name} PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME moonstone.${name} COMMAND moonstone_test_${name})
  endfunction()
  if(MOONSTONE_BUILD_CHISEL)
    moonstone_test(borrow chisel/tests/borrow.cpp moonstone_chisel)
    moonstone_test(gc chisel/tests/gc.cpp moonstone_chisel)
  endif()
  if(MOONSTONE_BUILD_SUPERSONIC)
    moonstone_test(synthesis supersonic/tests/synthesis.cpp moonstone_supersonic)
  endif()
  if(MOONSTONE_BUILD_SYREN)
    moonstone_test(catalog syren/tests/catalog.cpp moonstone_syren)
  endif()
  if(MOONSTONE_BUILD_EXOML)
    moonstone_test(exoml exoml/tests/interpreter.cpp moonstone_exoml)
    foreach(feature IN ITEMS autobind parser static_ffi)
      moonstone_test(exoml_${feature} exoml/tests/${feature}.cpp moonstone_exoml)
      set_tests_properties(moonstone.exoml_${feature} PROPERTIES TIMEOUT 30)
    endforeach()
  endif()
  if(MOONSTONE_BUILD_CHISEL AND MOONSTONE_BUILD_SYREN AND MOONSTONE_BUILD_SUPERSONIC AND MOONSTONE_BUILD_EXOML)
    moonstone_test(integration tests/integration.cpp moonstone_chisel)
    target_link_libraries(moonstone_test_integration PRIVATE moonstone_syren moonstone_supersonic moonstone_exoml)
  endif()
  if(MOONSTONE_BUILD_PLAID)
    foreach(feature IN ITEMS continuum cps_machine weaver loom conduit chorus parallax sentinel ir loom_stress conduit_stress chorus_stress parallax_failure)
      moonstone_test(plaid_${feature} plaid/tests/${feature}.cpp moonstone_plaid)
      set_tests_properties(moonstone.plaid_${feature} PROPERTIES TIMEOUT 60)
    endforeach()
    set_tests_properties(moonstone.plaid_loom_stress moonstone.plaid_conduit_stress moonstone.plaid_chorus_stress
      PROPERTIES LABELS "plaid;stress")
    moonstone_test(plaid_c_api plaid/tests/c_api.c moonstone_plaid_runtime)
    set_target_properties(moonstone_test_plaid_c_api PROPERTIES LINKER_LANGUAGE CXX C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)

    # Each public header is its own translation unit: umbrella includes must
    # not hide missing dependencies. This is a build-time check, not a proof.
    file(GLOB_RECURSE plaid_public_headers CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}"
      "${CMAKE_CURRENT_SOURCE_DIR}/plaid/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/plaid/*.h")
    set(plaid_header_sources)
    foreach(header IN LISTS plaid_public_headers)
      string(MAKE_C_IDENTIFIER "${header}" header_id)
      set(source "${CMAKE_CURRENT_BINARY_DIR}/plaid-header-checks/${header_id}.cpp")
      file(GENERATE OUTPUT "${source}" CONTENT "#include \"${header}\"\n")
      list(APPEND plaid_header_sources "${source}")
    endforeach()
    add_library(moonstone_plaid_header_checks OBJECT ${plaid_header_sources})
    target_link_libraries(moonstone_plaid_header_checks PRIVATE moonstone_plaid)
    if(MSVC)
      target_compile_options(moonstone_plaid_header_checks PRIVATE /W4 /WX)
    else()
      target_compile_options(moonstone_plaid_header_checks PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endif()
endif()

# Public headers are available from the source tree. Installation is scoped to
# the maintained header-only API and a separately built C runtime library.
install(DIRECTORY core/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/moonstone/core FILES_MATCHING PATTERN "*.hpp")
foreach(component IN ITEMS chisel syren supersonic exoml)
  if(TARGET moonstone_${component})
    install(DIRECTORY ${component}/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  endif()
endforeach()

  install(DIRECTORY plaid/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/moonstone/plaid FILES_MATCHING PATTERN "*.hpp" PATTERN "*.h")
