cmake_minimum_required(VERSION 3.20)
project(MoonstoneExternalConsumer LANGUAGES C CXX)
enable_testing()
find_package(Moonstone 0.1 CONFIG REQUIRED COMPONENTS Core)
if(MOONSTONE_TEST_MISSING_COMPONENT)
  find_package(Moonstone 0.1 CONFIG REQUIRED COMPONENTS NotAComponent)
endif()
add_executable(consumer_cpp main.cpp)
target_link_libraries(consumer_cpp PRIVATE Moonstone::Core)
add_executable(consumer_c main.c)
target_link_libraries(consumer_c PRIVATE Moonstone::Core)
foreach(component IN ITEMS Chisel Syren Supersonic Exoml)
  if(TARGET Moonstone::${component})
    target_link_libraries(consumer_cpp PRIVATE Moonstone::${component})
  endif()
endforeach()
add_test(NAME consumer.cpp COMMAND consumer_cpp)
add_test(NAME consumer.c COMMAND consumer_c)
