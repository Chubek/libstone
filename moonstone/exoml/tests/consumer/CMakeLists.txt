cmake_minimum_required(VERSION 3.20)
project(ExoMLInstalledConsumer LANGUAGES CXX)
find_package(MoonstoneExoML CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE Moonstone::ExoML)
if(TEST_EXTRACT)
  find_package(MoonstoneExoMLExtract CONFIG REQUIRED)
  target_link_libraries(consumer PRIVATE Moonstone::ExoMLExtract)
  target_compile_definitions(consumer PRIVATE TEST_EXTRACT
    EXTRACT_HEADER="${CMAKE_CURRENT_SOURCE_DIR}/discovery.h")
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_compile_definitions(consumer PRIVATE
      EXTRACT_CACHE_DIRECTORY="${CMAKE_CURRENT_BINARY_DIR}/extract-cache")
  endif()
endif()
enable_testing()
add_test(NAME installed_consumer COMMAND consumer)
