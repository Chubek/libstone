cmake_minimum_required(VERSION 3.20)
project(ArchBFDExamples LANGUAGES C)
option(ARCHBFD_EXAMPLE_C_ONLY "Build the shared-library example without a C++ compiler" OFF)
if(NOT ARCHBFD_EXAMPLE_C_ONLY)
  enable_language(CXX)
endif()
find_package(ArchBFD 1 CONFIG REQUIRED)
add_executable(archbfd-sections archbfd-sections.c)
target_link_libraries(archbfd-sections PRIVATE ArchBFD::archbfd)
if(NOT ARCHBFD_EXAMPLE_C_ONLY)
  set_target_properties(archbfd-sections PROPERTIES LINKER_LANGUAGE CXX)
  add_executable(archbfd-inspect archbfd-inspect.cpp)
  target_link_libraries(archbfd-inspect PRIVATE ArchBFD::cpp)
endif()
enable_testing()
add_test(NAME sections_self COMMAND archbfd-sections $<TARGET_FILE:archbfd-sections>)
if(TARGET archbfd-inspect)
  add_test(NAME inspect_self COMMAND archbfd-inspect $<TARGET_FILE:archbfd-inspect>)
endif()
