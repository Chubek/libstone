cmake_minimum_required(VERSION 3.20)
project(Weighstone VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
find_package(Boost REQUIRED CONFIG)
add_library(Weighstone INTERFACE)
add_library(Weighstone::Weighstone ALIAS Weighstone)
target_compile_features(Weighstone INTERFACE cxx_std_20)
target_include_directories(Weighstone INTERFACE
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(Weighstone INTERFACE Boost::headers)
add_executable(weighstone_basic examples/basic.cpp)
target_link_libraries(weighstone_basic PRIVATE Weighstone)
add_executable(weighstone_loop_tensor examples/loop_tensor.cpp)
target_link_libraries(weighstone_loop_tensor PRIVATE Weighstone)
if(BUILD_TESTING)
  add_executable(weighstone_tests tests/test_ccint.cpp)
  target_link_libraries(weighstone_tests PRIVATE Weighstone)
  add_test(NAME ccint COMMAND weighstone_tests)
  add_executable(weighstone_properties tests/test_properties.cpp)
  target_link_libraries(weighstone_properties PRIVATE Weighstone)
  add_test(NAME properties COMMAND weighstone_properties)
  add_executable(weighstone_analysis tests/test_analysis.cpp)
  target_link_libraries(weighstone_analysis PRIVATE Weighstone)
  add_test(NAME analysis COMMAND weighstone_analysis)
  add_executable(weighstone_optimization tests/test_optimization.cpp)
  target_link_libraries(weighstone_optimization PRIVATE Weighstone)
  add_test(NAME optimization COMMAND weighstone_optimization)
  add_test(NAME loop_tensor_example COMMAND weighstone_loop_tensor)
  # Compile every header in its own translation unit: umbrella include order
  # must not conceal missing dependencies in a header-only library.
  file(GLOB_RECURSE WEIGHSTONE_HEADERS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/include/*.hpp" "${CMAKE_CURRENT_SOURCE_DIR}/support/*.hpp")
  set(WEIGHSTONE_HEADER_CHECKS)
  foreach(header IN LISTS WEIGHSTONE_HEADERS)
    file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}" "${header}")
    string(MAKE_C_IDENTIFIER "${relative}" identifier)
    set(source "${CMAKE_CURRENT_BINARY_DIR}/header_checks/${identifier}.cpp")
    file(GENERATE OUTPUT "${source}" CONTENT "#include \"${relative}\"\n")
    list(APPEND WEIGHSTONE_HEADER_CHECKS "${source}")
  endforeach()
  add_library(weighstone_header_checks OBJECT ${WEIGHSTONE_HEADER_CHECKS})
  target_link_libraries(weighstone_header_checks PRIVATE Weighstone)
endif()
